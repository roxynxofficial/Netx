cmake_minimum_required(VERSION 3.20)
project(NETX VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_executable(netx src/main.cpp)
target_include_directories(netx PRIVATE include)
if(WIN32)
  target_link_libraries(netx PRIVATE ws2_32 bcrypt iphlpapi)
  target_compile_definitions(netx PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
else()
  target_link_libraries(netx PRIVATE pthread)
endif()
if(MSVC)
  target_compile_options(netx PRIVATE /W4 /permissive- /EHsc)
else()
  target_compile_options(netx PRIVATE -Wall -Wextra -Wpedantic)
endif()

enable_testing()
add_executable(netx_parser_test tests/parser_test.cpp)
target_include_directories(netx_parser_test PRIVATE include)
add_test(NAME parser COMMAND netx_parser_test)
